# PushPal — Service Worker Design

## Overview

The service worker handles push notification events, notification clicks, and PWA lifecycle events. It is generated by Next.js PWA tooling or manually maintained.

---

## Service Worker Events

### 1. Push Event

Receives push notifications from the push service and displays them to the user.

```javascript
// sw.js

self.addEventListener('push', (event) => {
  if (!event.data) return;

  const data = event.data.json();

  const options = {
    body: data.body || '',
    icon: '/icons/icon-192.png',
    badge: '/icons/badge-72.png',
    tag: data.tag || 'pushpal-notification',
    renotify: true,
    vibrate: [200, 100, 200],
    data: {
      notificationId: data.notificationId,
      url: data.url || '/dashboard'
    },
    actions: []  // MVP: no actions
  };

  event.waitUntil(
    self.registration.showNotification(data.title, options)
  );
});
```

### Push Payload Format

```json
{
  "title": "Take out the trash",
  "body": "Don't forget!",
  "tag": "notification-abc-123",
  "notificationId": "abc-123",
  "url": "/dashboard/abc-123",
  "icon": "/icons/icon-192.png",
  "badge": "/icons/badge-72.png"
}
```

### 2. Notification Click Event

Handles when user taps the notification.

```javascript
self.addEventListener('notificationclick', (event) => {
  event.notification.close();

  const notificationId = event.notification.data.notificationId;
  const url = event.notification.data.url || '/dashboard';

  event.waitUntil(
    clients.matchAll({ type: 'window', includeUncontrolled: true })
      .then((clientList) => {
        // Focus existing window if open
        for (const client of clientList) {
          if (client.url.includes(self.location.origin) && 'focus' in client) {
            client.navigate(url);
            return client.focus();
          }
        }

        // Open new window if none exists
        if (clients.openWindow) {
          return clients.openWindow(url);
        }
      })
  );

  // Mark notification as viewed (handled by frontend on load)
});
```

**Behavior:**
- If PushPal is already open → focus and navigate to notification
- If PushPal is not open → open new window to notification
- Always close the notification banner

### 3. Install Event

Fires when the PWA is being installed.

```javascript
self.addEventListener('install', (event) => {
  // Skip waiting to activate immediately
  self.skipWaiting();

  // Pre-cache static assets (optional)
  event.waitUntil(
    caches.open('pushpal-v1').then((cache) => {
      return cache.addAll([
        '/',
        '/icons/icon-192.png',
        '/icons/icon-512.png',
        '/icons/badge-72.png'
      ]);
    })
  );
});
```

---

### 4. Activate Event

Fires when the service worker takes control. Clean up old caches.

```javascript
self.addEventListener('activate', (event) => {
  // Clean up old caches
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames
          .filter((name) => name !== 'pushpal-v1')
          .map((name) => caches.delete(name))
      );
    })
  );

  // Claim all clients immediately
  self.clients.claim();
});
```

---

### 5. Fetch Event (Offline Support)

Cache static assets for offline access. API calls are not cached.

```javascript
self.addEventListener('fetch', (event) => {
  // Only cache GET requests
  if (event.request.method !== 'GET') return;

  // Don't cache API calls
  if (event.request.url.includes('/api/')) return;

  event.respondWith(
    caches.match(event.request).then((cachedResponse) => {
      // Return cached response if available
      if (cachedResponse) {
        return cachedResponse;
      }

      // Otherwise fetch from network
      return fetch(event.request).then((response) => {
        // Cache successful responses
        if (response.status === 200) {
          const responseClone = response.clone();
          caches.open('pushpal-v1').then((cache) => {
            cache.put(event.request, responseClone);
          });
        }
        return response;
      });
    })
  );
});
```

## Push Subscription Management

### Subscribe to Push

```typescript
// lib/push.ts

export async function subscribeToPush(): Promise<PushSubscription> {
  const registration = await navigator.serviceWorker.ready;

  const subscription = await registration.pushManager.subscribe({
    userVisibleOnly: true,
    applicationServerKey: urlBase64ToUint8Array(VAPID_PUBLIC_KEY)
  });

  return subscription;
}

export async function unsubscribeFromPush(): Promise<void> {
  const registration = await navigator.serviceWorker.ready;
  const subscription = await registration.pushManager.getSubscription();

  if (subscription) {
    await subscription.unsubscribe();
  }
}

export async function getExistingSubscription(): Promise<PushSubscription | null> {
  const registration = await navigator.serviceWorker.ready;
  return registration.pushManager.getSubscription();
}
```

### Send Subscription to Backend

```typescript
export async function registerSubscription(subscription: PushSubscription): Promise<void> {
  const keys = subscription.toJSON().keys;

  await api.post('/devices', {
    endpoint: subscription.endpoint,
    keys: {
      p256dh: keys?.p256dh,
      auth: keys?.auth
    }
  });
}
```

## Service Worker Registration

```typescript
// In Next.js app layout or page

if ('serviceWorker' in navigator) {
  window.addEventListener('load', () => {
    navigator.serviceWorker.register('/sw.js')
      .then((registration) => {
        console.log('SW registered:', registration.scope);
      })
      .catch((error) => {
        console.error('SW registration failed:', error);
      });
  });
}
```
